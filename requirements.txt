# Runtime dependencies for the application.
#
# Note what is NOT here: torch, torchvision, timm, albumentations. The app
# serves an exported ONNX graph, so the ~2.5 GB training stack is not needed to
# run it. Those live in requirements-train.txt, which is what the Kaggle
# notebook needs. Keeping them apart is what lets the container stay around
# 600 MB instead of 4 GB, and what lets a free hosting tier install this at all.

# Streamlit and FastAPI share Starlette, and their allowed ranges must overlap:
# streamlit 1.64 needs starlette>=0.46, which fastapi 0.115 forbids. These two
# pins resolve together on starlette 1.7 — check that before bumping either.

# --- web layer -------------------------------------------------------------
streamlit==1.64.0            # the Streamlit front end (streamlit_app.py)
fastapi==0.141.1             # the JSON API (backend/main.py)
uvicorn[standard]==0.34.0
python-multipart==0.0.20     # FastAPI file uploads
pydantic==2.10.4

# --- inference -------------------------------------------------------------
onnxruntime==1.20.1          # runs the exported backbone on CPU
numpy==2.1.3
opencv-python-headless==4.10.0.84   # headless: no GUI libs on a server
pandas==2.2.3
